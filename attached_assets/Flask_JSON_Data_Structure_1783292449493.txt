@app.get("/api/telemetry/latest")
def get_latest_greenhouse_metrics():
    """Consolidates rows across all three files into a clean JSON layout."""
    weather_row = get_last_csv_row(CSV_PATH_WEATHER)
    solar_row = get_last_csv_row(CSV_PATH_SOLAR)
    internal_row = get_last_csv_row(CSV_PATH_INTERNAL)
    
    payload_matrix = {
        "outdoor_weather": {},
        "solar_irradiance": {},
        "indoor_climat_hd50": {}
    }
    
    if weather_row:
        payload_matrix["outdoor_weather"] = {
            "timestamp": weather_row[0],
            "device_name": weather_row[1],
            "rssi": weather_row[3],
            "temperature_c": weather_row[5],
            "humidity_pct": weather_row[6],
            "wind_speed_kmh": weather_row[7],
            "wind_cardinal": weather_row[10],
            "rain_mm": weather_row[11]
        }
        
    if solar_row:
        payload_matrix["solar_irradiance"] = {
            "timestamp": solar_row[0],
            "device_name": solar_row[1],
            "battery_v": solar_row[5],
            "irradiance_wm2": solar_row[7]
        }
        
    if internal_row:
        payload_matrix["indoor_climat_hd50"] = {
            "timestamp": internal_row[0],
            "temperature_c": internal_row[1],
            "humidity_pct": internal_row[2],
            "dew_point_c": internal_row[3],
            "partial_vapor_pressure_hPa": internal_row[4],
            "illuminance_lux": internal_row[5],
            "co2_ppm": internal_row[6],
            "atmospheric_pressure_hPa": internal_row[7]
        }
        
    return payload_matrix
